Calculate the corrected concentrations based on the pump data and added ID column for PCA analysis and ug/m^3

dat <- dat_joined |>
  mutate(conc_ngcm = amount/cubic_m,
         conc_ugcm = conc_ngcm/1000,
         pca_id = case_when(

    type == "sample" ~ str_c(
      "sample", pump, time, location, season, year,
      sep = "_"
    ),

    type == "control" ~ str_c(
      "control", time, location, season, year,
      sep = "_"
    ),

    type == "blk" ~ str_c(
      blk_num, location, season, year,
      sep = "_"
    ),
    TRUE ~ NA_character_))
